   1  // build-3d.mjs — authored-asset pipeline (/lab.html "Deep Signal" +
   2  // /about.html "Signal Field v5 — Machined").
   3  // Blender (headless) → raw glb + matcap png → Draco glb + KTX2 texture
   4  // → src/assets/3d/ (committed; NOT part of npm run build — Blender and
   5  // toktx are machine deps, the outputs are checked in).
   6  //
   7  //   node scripts/build-3d.mjs            # all bundles incl. Blender
   8  //   node scripts/build-3d.mjs about      # one bundle by name
   9  //   node scripts/build-3d.mjs --no-bake  # reuse /tmp raw outputs
  10  //
  11  // Machine deps: /opt/homebrew/bin/blender (5.x), /opt/homebrew/bin/toktx
  12  // (KTX-Software, hand-installed — no brew formula), npx gltf-transform.
  13

  15  import { cpSync, mkdirSync, existsSync, statSync } from "node:fs";
  16  
  17  const BLENDER = "/opt/homebrew/bin/blender";
  18  const TOKTX = "/opt/homebrew/bin/toktx";
  19  const OUT = "src/assets/3d";

  21  // one entry per page bundle: Blender authoring script + its outputs
  22  const BUNDLES = {
  23    worlds: { script: "assets-src/worlds/gen_atlas.py", tmp: "/tmp/worlds-assets", glb: "worlds-atlas", matcap: null },
  24    // rdo: v3 matcaps are 1024² — UASTC RDO halves them (~840→~445 KB) with no visible loss on soft gradients
  25    lab: { script: "assets-src/lab/gen_crystals.py", tmp: "/tmp/lab-assets", glb: "lab-crystals", matcap: "lab-matcap", extraMatcaps: ["lab-matcap-int"], rdo: true },
  26    about: { script: "assets-src/about/gen_instrument.py", tmp: "/tmp/about-assets", glb: "about-instrument", matcap: "about-matcap" },
  27    // v6 ruin set-pieces: geometry only — reuses the lab matcaps at runtime

  29  };

  31  const run = (cmd, args) => {
  32    console.log(`[3d] ${cmd.split("/").pop()} ${args.join(" ")}`);
  33    execFileSync(cmd, args, { stdio: "inherit" });
  34  };
  35  const kb = (p) => `${(statSync(p).size / 1024).toFixed(1)} KB`;
  36  
  37  const only = process.argv.slice(2).filter((a) => !a.startsWith("--"));
  38  const picked = Object.entries(BUNDLES).filter(([name]) => !only.length || only.includes(name));
  39  if (!picked.length) { console.error(`[3d] unknown bundle "${only}" — have: ${Object.keys(BUNDLES).join(", ")}`); process.exit(1); }
  40  
  41  mkdirSync(`${OUT}/draco`, { recursive: true });
  42  mkdirSync(`${OUT}/basis`, { recursive: true });
  43  
  44  for (const [name, b] of picked) {
  45    mkdirSync(b.tmp, { recursive: true });
  46  
  47    // 1 · author in Blender (deterministic)
  48    if (!process.argv.includes("--no-bake")) {
  49      run(BLENDER, ["--background", "--factory-startup", "--python", b.script, "--", b.tmp]);
  50    }
  51    for (const f of [`${b.tmp}/${b.glb}-raw.glb`, ...(b.matcap ? [`${b.tmp}/${b.matcap}.png`] : [])]) {
  52      if (!existsSync(f)) { console.error(`[3d] missing ${f} — run without --no-bake`); process.exit(1); }
  53    }
  54  
  55    // 2 · Draco-compress the geometry
  56    run("npx", ["gltf-transform", "draco", `${b.tmp}/${b.glb}-raw.glb`, `${OUT}/${b.glb}.glb`]);
  57  
  58    // 3 · KTX2 the matcap(s) (UASTC — all soft gradients, ETC1S bands)
  59    for (const m of [...(b.matcap ? [b.matcap] : []), ...(b.extraMatcaps || [])]) {
  60      if (!existsSync(`${b.tmp}/${m}.png`)) { console.error(`[3d] missing ${b.tmp}/${m}.png — run without --no-bake`); process.exit(1); }
  61      run(TOKTX, ["--t2", "--encode", "uastc", "--uastc_quality", "3",
  62        ...(b.rdo ? ["--uastc_rdo_l", "1.5", "--uastc_rdo_d", "8192"] : []),
  63        "--zcmp", "19", "--genmipmap", "--assign_oetf", "srgb", `${OUT}/${m}.ktx2`, `${b.tmp}/${m}.png`]);
  64    }
  65  
  66    console.log(`[3d] ${name} ok — glb ${kb(`${OUT}/${b.glb}.glb`)}${b.matcap ? `, matcap ${kb(`${OUT}/${b.matcap}.ktx2`)}` : ""}`);
  67  }

  72    cpSync(`${LIBS}/draco/gltf/${f}`, `${OUT}/draco/${f}`);
  73  }
  74  for (const f of ["basis_transcoder.js", "basis_transcoder.wasm"]) {
  75    cpSync(`${LIBS}/basis/${f}`, `${OUT}/basis/${f}`);
