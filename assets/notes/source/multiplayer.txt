 176  function supaTransport(h) {
 177    let sb = null, ch = null;
 178    return {
 179      kind: "supa",
 180      join(code, meta) {
 181        const cfg = window.__gameNetCfg || {};   /* QA override hook (same pattern as __axonAuthCfg) */
 182        sb = createClient(cfg.url || SUPA_URL, cfg.key || SUPA_KEY, { realtime: { params: { eventsPerSecond: 32 } } });
 183        ch = sb.channel("strike-" + code, { config: { broadcast: { self: false, ack: false }, presence: { key: meta.id } } });
 184        ch.on("broadcast", { event: "g" }, (m) => h.onMsg(m.payload));
 185        ch.on("presence", { event: "sync" }, () => {
 186          const st = ch.presenceState(), peers = new Map();
 187          for (const k of Object.keys(st)) { if (k !== meta.id && st[k][0]) peers.set(k, st[k][0]); }
 188          h.onPeers(peers);
 189        });
 190        ch.subscribe((status) => {
 191          if (status === "SUBSCRIBED") { ch.track(meta).catch(() => {}); h.onState("on"); }
 192          else if (status === "CHANNEL_ERROR" || status === "TIMED_OUT" || status === "CLOSED") h.onState("err");
 193        });
 194      },
 195      send(p) { try { ch?.send({ type: "broadcast", event: "g", payload: p }); } catch (e) { /* transient */ } },
 196      leave() { try { if (ch) sb.removeChannel(ch); } catch (e) { /* closing */ } ch = null; h.onState("off"); },
 197    };
 198  }
